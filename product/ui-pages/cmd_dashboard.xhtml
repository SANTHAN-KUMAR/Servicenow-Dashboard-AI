var table = RP.getParameterValue('table') || 'incident'; var rawPath = RP.getParameterValue('path') || ''; /* The drill path travels in the URL as field:key|field:key, so a drill state is shareable and the browser back button reverses it for free. */ var path = []; if (rawPath) { var segs = rawPath.split('|'); for (var i = 0; i < segs.length; i++) { var bits = segs[i].split(':'); if (!bits[0]) continue; path.push({ field: decodeURIComponent(bits[0]), key: bits.length > 1 ? decodeURIComponent(bits[1]) : '' }); } } /* The time window, as a slicer in the URL.
